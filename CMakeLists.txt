cmake_minimum_required(VERSION 3.21)
project(DesktopSnow VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(APPLE OR WIN32)
  set(SNOW_DESKTOP_DEFAULT ON)
else()
  set(SNOW_DESKTOP_DEFAULT OFF)
endif()
option(SNOW_BUILD_DESKTOP "Build the macOS / Windows desktop overlay" ${SNOW_DESKTOP_DEFAULT})
option(SNOW_BUILD_TESTS "Build deterministic simulation tests" ON)
option(SNOW_ENABLE_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)

add_library(snow_engine STATIC src/engine.cpp)
target_include_directories(snow_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(snow_engine PRIVATE /W4 /permissive-)
else()
  target_compile_options(snow_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(SNOW_ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(snow_engine PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(snow_engine PRIVATE -fsanitize=address,undefined)
endif()

if(SNOW_BUILD_TESTS)
  include(CTest)
  add_executable(snow_tests tests/tests.cpp)
  target_link_libraries(snow_tests PRIVATE snow_engine)
  add_test(NAME SnowEngine COMMAND snow_tests)
  add_executable(snow_benchmark tests/benchmark.cpp)
  target_link_libraries(snow_benchmark PRIVATE snow_engine)
endif()

if(SNOW_BUILD_DESKTOP)
  if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "The desktop overlay currently targets only macOS and Windows")
  endif()
  include(FetchContent)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(glfw URL https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(glfw)
  add_executable(desktop_snow src/main.cpp src/renderer.cpp)
  target_include_directories(desktop_snow PRIVATE include)
  target_include_directories(desktop_snow PRIVATE src)
  target_link_libraries(desktop_snow PRIVATE snow_engine glfw)
  if(APPLE)
    enable_language(OBJCXX)
    target_sources(desktop_snow PRIVATE src/platform_mac.mm)
    target_link_libraries(desktop_snow PRIVATE "-framework AppKit" "-framework CoreGraphics" "-framework Foundation")
    set_target_properties(desktop_snow PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_INFO_PLIST ${CMAKE_CURRENT_SOURCE_DIR}/src/Info.plist
      XCODE_ATTRIBUTE_MACOSX_DEPLOYMENT_TARGET "12.0")
    target_compile_options(desktop_snow PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
  elseif(WIN32)
    set_target_properties(desktop_snow PROPERTIES WIN32_EXECUTABLE TRUE)
    target_sources(desktop_snow PRIVATE src/platform_win.cpp src/app.rc)
    target_link_libraries(desktop_snow PRIVATE user32 gdi32 shell32 dwmapi ole32 uuid)
    target_compile_definitions(desktop_snow PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  endif()
endif()
